cmake_minimum_required(VERSION 3.22.1)
project(mote_vlm LANGUAGES C CXX ASM)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized local inference" FORCE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(MOTE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE)
set(LLAMA_ROOT "${MOTE_ROOT}/vendor/llama.cpp" CACHE PATH "Pinned llama.cpp")
if(NOT EXISTS "${LLAMA_ROOT}/tools/mtmd/mtmd.h")
  message(FATAL_ERROR "Missing pinned vendor/llama.cpp; run the Mote native setup first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_OPENCL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "K90 / arm64 fixture target" FORCE)
add_subdirectory("${LLAMA_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(mote_vlm SHARED native_vlm.cpp)
target_include_directories(mote_vlm PRIVATE "${LLAMA_ROOT}/vendor")
target_compile_options(mote_vlm PRIVATE -Wall -Wextra -Werror=return-type -fvisibility=hidden -fno-fast-math)
target_link_options(mote_vlm PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
target_link_libraries(mote_vlm PRIVATE llama mtmd log android dl)
